import { ACQUIRERS, CAPABILITY_LABELS, DISPLAY_ORDER, PATH_LABELS, CARD_PRESENT_PATHS } from '@site/src/data/acquirerCaps';

export function AcquirerMatrix() {
  // Paths to show as columns (card-present + back-office)
  const paths = [...CARD_PRESENT_PATHS, 'back-office'];

  return (
    <div style={{ overflowX: 'auto' }}>
      {ACQUIRERS.map(acq => {
        // Only show rows that have at least one supported path
        const rows = DISPLAY_ORDER.filter(cap =>
          acq.caps[cap] && paths.some(p => acq.caps[cap][p] === 'public')
        );
        if (rows.length === 0) return null;
        return (
          <div key={acq.id} style={{ marginBottom: '2rem' }}>
            <h3 id={acq.id}>{acq.name} <small style={{ fontWeight: 400, color: 'var(--ifm-color-emphasis-600)' }}>— {acq.subtitle}</small></h3>
            <table>
              <thead>
                <tr>
                  <th>Capability</th>
                  {paths.map(p => <th key={p}>{PATH_LABELS[p] || p}</th>)}
                </tr>
              </thead>
              <tbody>
                {rows.map(cap => (
                  <tr key={cap}>
                    <td>{CAPABILITY_LABELS[cap] || cap}</td>
                    {paths.map(p => {
                      const v = (acq.caps[cap] || {})[p];
                      return <td key={p} style={{ textAlign: 'center' }}>{v === 'public' ? '✅' : v === 'not-supported' ? '❌' : '—'}</td>;
                    })}
                  </tr>
                ))}
              </tbody>
            </table>
            {acq.portalNote && (
              <p style={{ fontSize: '0.85rem', color: 'var(--ifm-color-emphasis-700)', marginTop: '0.5rem' }}>
                <strong>Paysafe Portal note:</strong> {acq.portalNote}
              </p>
            )}
          </div>
        );
      })}
    </div>
  );
}
